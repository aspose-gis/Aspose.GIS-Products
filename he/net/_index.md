        Console.WriteLine(feature.GetValue&lt;string&gt;("CITY_NAME")); // הדפיסו ערך תכונה

        Console.WriteLine(feature.GetValue&lt;DateTime&gt;("UPDATE_DAT")); 
    }

}</code></pre>
